local Log = {
    lines = {},
    max = 600,
    verbose = false,
}

function Log.write(level, section, message)
    local entry = string.format("[%7.2f][%-5s][%-10s] %s",
        os.clock(), level, section, tostring(message))
    Log.lines[#Log.lines + 1] = entry
    if #Log.lines > Log.max then
        table.remove(Log.lines, 1)
    end
    if Log.verbose then
        print(entry)
    end
    return entry
end

function Log.info(section, message) return Log.write("INFO", section, message) end
function Log.warn(section, message) return Log.write("WARN", section, message) end
function Log.err(section, message) return Log.write("ERROR", section, message) end

function Log.probe(section, fn)
    local ok, result = pcall(fn)
    if ok then
        Log.info(section, result)
    else
        Log.err(section, result)
    end
    return ok, result
end

local function loadLib(path)
    if type(loadstring) ~= "function" then
        return nil, "loadstring unavailable"
    end
    local okHttp, source = pcall(function()
        return game:HttpGet(BASE .. path)
    end)
    if not okHttp or type(source) ~= "string" then
        return nil, "HttpGet failed"
    end
    local okCompile, chunk = pcall(loadstring, source)
    if not okCompile or type(chunk) ~= "function" then
        return nil, "compile failed: " .. tostring(chunk)
    end
    local okRun, value = pcall(chunk)
    if not okRun or type(value) ~= "table" then
        return nil, "module error: " .. tostring(value)
    end
    return value
end

local function attr(inst, name)
    if typeof(inst) ~= "Instance" then
        return nil
    end
    local ok, value = pcall(inst.GetAttribute, inst, name)
    return ok and value or nil
end

local function rootOf(model)
    if typeof(model) ~= "Instance" then
        return nil
    end
    return model:FindFirstChild("HumanoidRootPart")
        or model:FindFirstChild("UpperTorso")
        or model:FindFirstChild("Torso")
        or model.PrimaryPart
        or model:FindFirstChildWhichIsA("BasePart")
end

local function isRig(model)
    return typeof(model) == "Instance"
        and model:IsA("Model")
        and model:FindFirstChild("HumanoidRootPart") ~= nil
end

local Rigs = {
    all = {},
    players = {},
    zombies = {},
    localRig = nil,
    lastScan = 0,
    roots = {},
}

local function candidateRoots()
    local roots = {}
    local ents = entitiesFolder()
    if ents then
        roots[#roots + 1] = ents
    end
    local chars = Workspace:FindFirstChild("Characters")
    if chars then
        roots[#roots + 1] = chars
    end
    local living = Workspace:FindFirstChild("Living")
    if living then
        roots[#roots + 1] = living
    end
    if #roots == 0 then
        roots[#roots + 1] = Workspace
    end
    return roots
end

local R15_PARTS = {
    UpperTorso = true,
    LowerTorso = true,
    RightUpperArm = true,
    LeftUpperArm = true,
    RightUpperLeg = true,
    LeftUpperLeg = true,
    RightLowerArm = true,
    LeftLowerArm = true,
}

local function looksLikePlayerRig(model)
    for _, c in ipairs(model:GetChildren()) do
        if R15_PARTS[c.Name] then
            return true
        end
    end
    if model:FindFirstChild("Torso") and model:FindFirstChild("Left Arm") then
        return false
    end
    return model:FindFirstChildOfClass("Humanoid") ~= nil
end

local function scanRigs(force)
    local now = os.clock()
    if not force and (now - Rigs.lastScan) < 0.5 then
        return Rigs
    end
    Rigs.lastScan = now

    local all, players, zombies = {}, {}, {}
    for _, root in ipairs(candidateRoots()) do
        for _, model in ipairs(root:GetChildren()) do
            if isRig(model) then
                all[#all + 1] = model
                if looksLikePlayerRig(model) then
                    players[#players + 1] = model
                else
                    zombies[#zombies + 1] = model
                end
            end
        end
    end

    local cam = Workspace.CurrentCamera
    local localRig, bestD, secondD
    if cam then
        local origin = cam.CFrame.Position
        for _, m in ipairs(all) do
            local r = rootOf(m)
            if r then
                local d = (r.Position - origin).Magnitude
                if not bestD or d < bestD then
                    localRig, secondD, bestD = m, bestD, d
                elseif not secondD or d < secondD then
                    secondD = d
                end
            end
        end
        local plausible = bestD and bestD < 40
        local isolated = (not secondD) or (secondD - (bestD or 0)) > 6
        if not (plausible and isolated) then
            localRig = nil
        end
    end
    Rigs.localDist = bestD

    local function strip(list)
        if not localRig then
            return list
        end
        local out = {}
        for _, m in ipairs(list) do
            if m ~= localRig then
                out[#out + 1] = m
            end
        end
        return out
    end

    Rigs.all = all
    Rigs.players = strip(players)
    Rigs.zombies = strip(zombies)
    Rigs.localRig = localRig
    return Rigs
end

local function rigHealth(model)
    local h = attr(model, "Health")
    if h then
        return h, attr(model, "MaxHealth") or 100
    end
    local states = model:FindFirstChild("States")
    if states then
        local sh = attr(states, "Health")
        if sh then
            return sh, attr(states, "MaxHealth") or 100
        end
    end
    local hum = model:FindFirstChildOfClass("Humanoid")
    if hum then
        return hum.Health, hum.MaxHealth
    end
    return 1, 1
end

local LootCache = {}
local function lootModels()
    local now = os.clock()
    if LootCache.t and (now - LootCache.t) < 1.0 then
        return LootCache.v
    end
    LootCache.t = now
    local ItemSpawns = itemSpawnsFolder()
    if not ItemSpawns then
        LootCache.v = {}
        return {}
    end
    local out = {}
    local want = State.lootRarity
    for _, item in ipairs(ItemSpawns:GetDescendants()) do
        if item:IsA("Model") then
            local display = attr(item, "DisplayName")
            if display then
                local rarity = attr(item, "Rarity") or "Common"
                if want == "All" or rarity == want then
                    out[#out + 1] = item
                end
            end
        end
    end
    LootCache.v = out
    return out
end

local function namedWorkspaceModels(name, requireChild)
    local out = {}
    for _, model in ipairs(Workspace:GetChildren()) do
        if model:IsA("Model") and model.Name == name then
            if not requireChild or model:FindFirstChild(requireChild) then
                out[#out + 1] = model
            end
        end
    end
    return out
end

local function isAirdrop(model)
    return model:GetAttribute("DropTimestamp") ~= nil
        or model:FindFirstChild("DropAnimator") ~= nil
end

local function airdropModels()
    local out = {}
    for _, model in ipairs(Workspace:GetChildren()) do
        if model:IsA("Model") and model.Name == "Container" and isAirdrop(model) then
            out[#out + 1] = model
        end
    end
    return out
end

local function lootContainerModels()
    local out = {}
    local function consider(model)
        if model:IsA("Model") and not isAirdrop(model) then
            local n = model.Name
            if n == "Container" or n:find("Locker") or n:find("Crate")
               or n:find("Cabinet") or n:find("Safe") or n:find("Barrel") then
                out[#out + 1] = model
            end
        end
    end
    for _, model in ipairs(Workspace:GetChildren()) do
        consider(model)
    end
    local spawns = itemSpawnsFolder()
    if spawns then
        for _, model in ipairs(spawns:GetDescendants()) do
            consider(model)
        end
    end
    return out
end

local function containerContents(model)
    local names = {}
    for _, d in ipairs(model:GetDescendants()) do
        local display = attr(d, "DisplayName")
        if display then
            names[#names + 1] = tostring(display)
            if #names >= 6 then
                break
            end
        end
    end
    return names
end

local function buildAdapter()
    local default = EasyAim.Adapters.Default
    return {
        targets = function()
            return {}
        end,
        model = function(raw)
            return raw
        end,
        root = default.root,
        part = default.part,
        health = function(model, raw)
            return rigHealth(model)
        end,
        alive = function(model, raw, health, maxHealth)
            if attr(model, "Dead") == true then
                return false
            end
            if maxHealth and maxHealth > 0 then
                return health > 0
            end
            return true
        end,
        name = function(model, raw)
            return model.Name
        end,
        team = function(model, raw)
            return attr(model, "SquadName")
        end,
        ignore = function(model, raw)
            return model == Rigs.localRig
        end,
    }
end

local function rigSource(playersFlag)

        local out = {}
        for _, m in ipairs(list) do
            out[#out + 1] = m
        end
        return out
    end
end

    local inst = ESP.new()

local function setupESP()
    if type(ESP) ~= "table" or type(ESP.new) ~= "function" then
        Log.err("esp", "library unavailable")
        return false
    end

    esp = newRosterESP(Color3.fromRGB(190, 150, 255), function()
        if not State.playerOn then
            return {}
        end
        return scanRigs().players
    end, function(model)
        local hp = rigHealth(model)
        return string.format("Player [%d]", math.floor((hp or 0) + 0.5))
    end)

    espZombie = newRosterESP(Color3.fromRGB(255, 135, 135), function()
        if not State.zombieOn then
            return {}
        end
        return scanRigs().zombies
    end, function(model)
        local hp = rigHealth(model)
        return string.format("Zombie [%d]", math.floor((hp or 0) + 0.5))
    end)

    local function group(id, getter, labeler, colour, priority)
        local spec = esp:addEnt(id, {
            get = getter,
            label = labeler,
            col = colour,
            box = true,
            name = true,
            dist = true,
            max = 5000,
            priority = priority,
        })
        entitySpecs[id] = spec
        return spec
    end

    group("loot", function()
        return State.lootOn and lootModels() or {}
    end, function(obj)
        local name = attr(obj, "DisplayName") or obj.Name
        local rarity = attr(obj, "Rarity")
        if rarity and rarity ~= "Common" then
            return name .. " [" .. rarity .. "]"
        end
        return name
    end, Color3.fromRGB(120, 255, 190), 2)

    group("weapons", function()
        return State.weaponOn and folderChildren(groundWeaponsFolder()) or {}
    end, function(obj)
        return attr(obj, "DisplayName") or obj.Name
    end, Color3.fromRGB(255, 220, 110), 2)

    group("ammo", function()
        return State.ammoOn and folderChildren(refillAmmoFolder()) or {}
    end, function()
        return "Ammo"
    end, Color3.fromRGB(255, 180, 120), 2)

    group("airdrops", function()
        return State.airdropOn and airdropModels() or {}
    end, function()
        return "Airdrop"
    end, Color3.fromRGB(255, 140, 90), 3)

    group("containers", function()
        return State.containerOn and lootContainerModels() or {}
    end, function(obj)
        if not State.peekContainers then
            return "Container"
        end
        local items = containerContents(obj)
        if #items == 0 then
            return "Container (empty)"
        end
        return "Container: " .. table.concat(items, ", ")
    end, Color3.fromRGB(200, 190, 120), 3)

    group("vehicles", function()
        return State.vehicleOn and vehicleModels() or {}
    end, function(obj)
        local states = obj:FindFirstChild("States")
        local hp = states and attr(states, "Health")
        return hp and string.format("Vehicle [%d]", math.floor(hp + 0.5)) or "Vehicle"
    end, Color3.fromRGB(160, 200, 255), 3)

    group("corpses", function()
        return State.corpseOn and corpseModels() or {}
    end, function()
        return "Body"
    end, Color3.fromRGB(200, 160, 160), 2)

    if POIFolder then
        group("poi", function()
            if not State.poiOn then
                return {}
            end
            local out = {}
            for _, poi in ipairs(POIFolder:GetChildren()) do
                local pos = attr(poi, "Position")
                if typeof(pos) == "Vector3" then
                    out[#out + 1] = pos
                end
            end
            return out
        end, false, Color3.fromRGB(200, 200, 210), 1)
        entitySpecs.poi.box = false
        entitySpecs.poi.name = false
    end

    Log.info("esp", "player + zombie instances started")
    return true
end

local function hardUnload()
    if Session.unloaded then
        return
    end
    Session.unloaded = true

    for _, conn in ipairs(Session.conns) do
        pcall(function() conn:Disconnect() end)
    end
    Session.conns = {}

    for _, getter in ipairs(Session.instances) do
        local inst = getter()
        if inst then
            pcall(function() inst:on(false) end)
            pcall(function() inst:stop() end)
            pcall(function() inst:destroy() end)
        end
    end

    if cleardrawcache then
        pcall(cleardrawcache)
    end

    local l = Session.restore.lighting
    if l then
        pcall(function()
            Lighting.Brightness = l.Brightness
            Lighting.ClockTime = l.ClockTime
            Lighting.FogEnd = l.FogEnd
            Lighting.Ambient = l.Ambient
            Lighting.OutdoorAmbient = l.OutdoorAmbient
        end)
    end

    pcall(function()
        local cam = Workspace.CurrentCamera
        if cam and Session.restore.fov then
            cam.FieldOfView = Session.restore.fov
        end
    end)

    pcall(function()
        LP.CameraMaxZoomDistance = 0.5
        LP.CameraMinZoomDistance = 0.5
    end)

    if GENV[SLOT] == Session then
        GENV[SLOT] = nil
    end

    GENV.AftermathUnload = nil
    Log.info("session", "unloaded")
end

local activeSlotCache = {}
local function findGunSlots(root)
    local out = {}
    if not root then return out end
    local ok, desc = pcall(root.GetDescendants, root)
    if not ok or type(desc) ~= "table" then return out end
    for _, d in ipairs(desc) do
        if d:IsA("ObjectValue") and d.Name == "InventorySlot" then
            out[#out + 1] = d
        end
    end
    return out
end

local function activeGunSlot()
    local now = os.clock()
    if activeSlotCache.t and (now - activeSlotCache.t) < 0.5 then
        return activeSlotCache.v
    end
    activeSlotCache.t = now
    for _, slot in ipairs(findGunSlots(LP)) do
        if slot.Value ~= nil then
            activeSlotCache.v = slot
            return slot
        end
    end
    activeSlotCache.v = nil
    return nil
end

local lastFire = 0
local function tryRageFire(inst, playersFlag)
    if not inst or not aimReady then return end
    if State.magicBulletOn == false and inst.target == nil then return end
    local now = os.clock()
    if (now - lastFire) < 0.05 then return end
    local origin = Workspace.CurrentCamera and Workspace.CurrentCamera.CFrame.Position
    if not origin then return end

    local pos, part
    if State.magicBulletOn or State.bulletTpOn then
        local rec = inst:pick()
        if not rec then
            return
        end
        pos, part = inst:resolve(rec)
    else
        local dir, hitPos = inst:aimDirection(origin)
        if not dir then
            return
        end
        pos = hitPos
        part = inst.hitPart and inst:hitPart(inst.target) or nil
    end

    if not pos then
        return
    end

    local target = part and part.Parent or (inst.target and inst.target.model)
    if not target or target == Rigs.localRig then
        return
    end

    local damageRemote = gunRemote(playersFlag and "DamagePlayer" or "DamageCharacter")
    if not damageRemote then
        return
    end

    lastFire = now

    local originPos = origin
    if State.bulletTpOn then
        originPos = pos + Vector3.new(0, 0, 0.01)
    end

    local ok, err = pcall(function()
        damageRemote:Fire(target, originPos, pos, {
            isPlayer = playersFlag,
            partName = part and part.Name or "Head",
            model = target,
        })
    end)
    if ok then
        Log.info("rage", "fire -> " .. target.Name)
    else
        Log.err("rage", err)
    end
end

local function inventoryItems()
    local out = {}
    local inv = LP:FindFirstChild("Inventory") or LP:FindFirstChild("Backpack")
    if not inv then return out end
    local ok, children = pcall(inv.GetChildren, inv)
    if not ok then return out end
    for _, c in ipairs(children) do
        out[#out + 1] = c
    end
    return out
end

local function useItemByName(match)
    for _, item in ipairs(inventoryItems()) do
        local display = attr(item, "DisplayName") or item.Name
        if display and display:lower():find(match) then
            local useFn = item:FindFirstChild("Use") or item:FindFirstChild("UseItem")
            if useFn and useFn:IsA("RemoteEvent") then
                pcall(function() useFn:FireServer() end)
                Log.info("survive", "used " .. tostring(display))
                return true
            end
        end
    end
    return false
end

local function buildReport()
    local rigs = scanRigs(true)
    local out = {}
    local function add(s) out[#out + 1] = s end

    add("===== AFTERMATH VERBOSE REPORT =====")
    add("version      : " .. VERSION)
    add("time         : " .. os.date("%Y-%m-%d %H:%M:%S"))
    add("executor     : " .. tostring(identifyexecutor and select(1, identifyexecutor()) or "?"))
    add("placeId      : " .. tostring(game.PlaceId))
    add("jobId        : " .. tostring(game.JobId))
    add("espReady     : " .. tostring(espReady))
    add("aimReady     : " .. tostring(aimReady))
    add("")

    add("--- ROSTER ---")
    add("Players service  : " .. #Players:GetPlayers())
    add("rigs total       : " .. #rigs.all)
    add("rigs -> players  : " .. #rigs.players)
    add("rigs -> zombies  : " .. #rigs.zombies)
    add("local rig        : " .. tostring(rigs.localRig and rigs.localRig.Name or "NOT FOUND"))
    add("nearest rig dist : " .. tostring(rigs.localDist and string.format("%.1f", rigs.localDist) or "?"))
    add("LP.Character     : " .. tostring(LP.Character and LP.Character.Name or "nil"))
    add("scan roots       : ")
    for _, r in ipairs(candidateRoots()) do
        add("    " .. r:GetFullName() .. "  (" .. #r:GetChildren() .. " children)")
    end
    add("")

    add("--- ESP STATE ---")
    if esp then
        add("cfg.on          : " .. tostring(esp.cfg.on))
        add("npc.on          : " .. tostring(esp.cfg.npc.on))
        add("npc.box.on      : " .. tostring(esp.cfg.npc.box.on))
        add("npc.espRange    : " .. tostring(esp.cfg.npc.espRange))
        local ok, stats = pcall(function() return esp:statsGet() end)
        if ok and type(stats) == "table" then
            add(string.format("drawn=%s pool=%s roster=%s ents=%s fps=%s",
                tostring(stats.drawn), tostring(stats.pool), tostring(stats.roster),
                tostring(stats.ents), tostring(stats.fps)))
        end
        local srcOk, srcCount = pcall(function()
            return #(esp._npcSrc and esp._npcSrc() or {})
        end)
        add("npc source returns : " .. (srcOk and tostring(srcCount) or "ERROR"))
    else
        add("esp is nil")
    end
    add("")

    add("--- AIM STATE ---")
    for name, inst in pairs({ legit = aimLegit, silentPlayers = aimSilentPlayers, silentZombies = aimSilentZombies }) do
        if inst then
            add(name .. ".on      : " .. tostring(inst.cfg.on))
            add(name .. ".mode    : " .. tostring(inst.cfg.mode))
            add(name .. ".running : " .. tostring(inst.running))
            local okRoster, roster = pcall(function() return #inst:build() end)
            add(name .. ".roster  : " .. (okRoster and tostring(roster) or "ERROR"))
        end
    end
    add("")

    add("--- WORLD SOURCES ---")
    add("loot        : " .. #lootModels())
    add("weapons     : " .. #folderChildren(groundWeaponsFolder()))
    add("ammo        : " .. #folderChildren(refillAmmoFolder()))
    add("containers  : " .. #lootContainerModels())
    add("airdrops    : " .. #airdropModels())
    add("vehicles    : " .. #vehicleModels())
    add("corpses     : " .. #corpseModels())
    add("")

    add("--- TOGGLES ---")
    for k, v in pairs(State) do
        add(string.format("  %-20s = %s", k, tostring(v)))
    end
    add("")

    add("--- LOG (" .. #Log.lines .. " lines) ---")
    for _, line in ipairs(Log.lines) do
        add(line)
    end

    return table.concat(out, "\n")
end

function Session.unload()
    for _, conn in ipairs(Session.conns) do
        pcall(function() conn:Disconnect() end)
    end
    Session.conns = {}
    for _, inst in ipairs({ aimLegit, aimSilentPlayers, aimSilentZombies, esp, espZombie }) do
        if inst then
            pcall(function() inst:destroy() end)
        end
    end
    local l = Session.restore.lighting
    if l then
        pcall(function()
            Lighting.Brightness = l.Brightness
            Lighting.ClockTime = l.ClockTime
            Lighting.FogEnd = l.FogEnd
            Lighting.Ambient = l.Ambient
            Lighting.OutdoorAmbient = l.OutdoorAmbient
        end)
    end
    pcall(function()
        local cam = Workspace.CurrentCamera
        if cam and Session.restore.fov then
            cam.FieldOfView = Session.restore.fov
        end
    end)
    pcall(function()
        LP.CameraMaxZoomDistance = 0.5
        LP.CameraMinZoomDistance = 0.5
    end)
    pcall(function() M:Close() end)
    if GENV[SLOT] == Session then
        GENV[SLOT] = nil
    end
end

local Visual = {}

function Visual.applyLighting()
    if not State.fullbright then
        return
    end
    if Lighting.Brightness ~= 3 then Lighting.Brightness = 3 end
    if Lighting.ClockTime ~= 14 then Lighting.ClockTime = 14 end
    local grey = Color3.fromRGB(178, 178, 178)
    if Lighting.Ambient ~= grey then Lighting.Ambient = grey end
    if Lighting.OutdoorAmbient ~= grey then Lighting.OutdoorAmbient = grey end
end

function Visual.applyFog()
    if State.noFog and Lighting.FogEnd < 1e5 then
        Lighting.FogEnd = 1e6
    end
end

function Visual.applyFov()
    if not State.fovOverride then
        return
    end
    local cam = Workspace.CurrentCamera
    if cam and math.abs(cam.FieldOfView - State.fovValue) > 0.25 then
        cam.FieldOfView = State.fovValue
    end
end

for _, prop in ipairs({ "Brightness", "ClockTime", "Ambient", "OutdoorAmbient" }) do
    Session.track(Lighting:GetPropertyChangedSignal(prop):Connect(Visual.applyLighting))
end
Session.track(Lighting:GetPropertyChangedSignal("FogEnd"):Connect(Visual.applyFog))

local function bindCameraFov()
    local cam = Workspace.CurrentCamera
    if cam then
        Session.track(cam:GetPropertyChangedSignal("FieldOfView"):Connect(Visual.applyFov))
    end
end
bindCameraFov()
Session.track(Workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(bindCameraFov))

local lastSelfTick = 0
local tickSignal = RunService.Stepped or RunService.Heartbeat or RunService.RenderStepped
Session.track(tickSignal:Connect(function(dt)
    local now = os.clock()
    if (now - lastSelfTick) < 0.033 then
        return
    end
    lastSelfTick = now

    if State.infStaminaOn then
        pcall(function()
            LP:SetAttribute("Stamina", attr(LP, "MaxStamina") or 100)
        end)
    end

    if State.thirdPersonOn then
        pcall(function()
            LP.CameraMaxZoomDistance = State.thirdPersonDist
            LP.CameraMinZoomDistance = State.thirdPersonDist
            LP.CameraMode = Enum.CameraMode.Classic
        end)
    else
        pcall(function()
            LP.CameraMaxZoomDistance = 0.5
            LP.CameraMinZoomDistance = 0.5
        end)
    end

    local hum = Movement.humanoid()
    if hum then
        if State.speedOn and hum.WalkSpeed ~= State.speedValue then
            hum.WalkSpeed = State.speedValue
        end
        if State.jumpOn and hum.JumpPower ~= State.jumpValue then
            hum.JumpPower = State.jumpValue
        end
    end

    if State.flyOn then
        local root = Movement.root()
        if root then
            local cam = Workspace.CurrentCamera
            local dir = Vector3.zero
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= cam.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += cam.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += cam.CFrame.UpVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= cam.CFrame.UpVector end
            local mag = dir.Magnitude
            if mag > 0 then
                dir = dir.Unit * State.flySpeed
            end
            root.AssemblyLinearVelocity = dir
        end
    end

    if State.autoEatOn and (now - (Movement.lastEat or 0)) > 4 then
        local hunger = attr(LP, "Hunger")
        local maxH = attr(LP, "MaxHunger") or 100
        if hunger and hunger < maxH * 0.6 then
            if useItemByName("food") or useItemByName("bread") or useItemByName("canned")
               or useItemByName("mre") or useItemByName("apple") or useItemByName("meat") then
                Movement.lastEat = now
            end
        end
    end
    if State.autoDrinkOn and (now - (Movement.lastDrink or 0)) > 4 then
        local thirst = attr(LP, "Thirst")
        local maxT = attr(LP, "MaxThirst") or 100
        if thirst and thirst < maxT * 0.6 then
            if useItemByName("water") or useItemByName("bottle") or useItemByName("drink")
               or useItemByName("soda") or useItemByName("canteen") then
                Movement.lastDrink = now
            end
        end
    end
    if State.autoHealOn and (now - (Movement.lastHeal or 0)) > 5 then
        local hp = attr(LP, "Health")
        local maxHp = attr(LP, "MaxHealth") or 100
        if hp and hp < maxHp * 0.7 then
            if useItemByName("bandage") or useItemByName("medkit") or useItemByName("medicine")
               or useItemByName("heal") or useItemByName("rags") then
                Movement.lastHeal = now
            end
        end
    end

    if State.infAmmoOn then
        local slot = activeGunSlot()
        if slot then
            local mag = slot:FindFirstChild("BulletsInMagazine")
            local reserve = slot:FindFirstChild("BulletsInReserve")
            if mag and mag.Value < 30 then
                mag.Value = 30
            end
            if reserve and reserve.Value < 300 then
                reserve.Value = 300
            end
        end
    end

    if State.rapidFireOn then
        pcall(function()
            local now2 = os.clock()
            if (now2 - (Movement.lastRapid or 0)) >= State.rapidFireRate then
                Movement.lastRapid = now2
                if mouse1press then
                    mouse1press()
                    task.delay(0.03, function()
                        if mouse1release then
                            mouse1release()
                        end
                    end)
                end
            end
        end)
    end

    if aimReady then
        if State.silentPlayersOn then
            tryRageFire(aimSilentPlayers, true)
        end
        if State.silentZombiesOn then
            tryRageFire(aimSilentZombies, false)
        end
    end
end))
