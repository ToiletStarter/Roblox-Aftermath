local function loadLib(path)
    if type(loadstring) ~= "function" then
        return nil, "loadstring unavailable"
    end
    local okHttp, source = pcall(function()
        return game:HttpGet(BASE .. path)
    end)
    if not okHttp or type(source) ~= "string" then
        return nil, "HttpGet failed"
    end
    local okCompile, chunk = pcall(loadstring, source)
    if not okCompile or type(chunk) ~= "function" then
        return nil, "compile failed"
    end
    local okRun, value = pcall(chunk)
    if not okRun or type(value) ~= "table" then
        return nil, "module returned " .. typeof(value)
    end
    return value
end

local GENV = (getgenv and getgenv()) or shared or _G
local SLOT = "__AFTERMATH_INSTANCE"

local function teardownPrevious()
    local prev = GENV[SLOT]
    if type(prev) ~= "table" then
        return false
    end
    GENV[SLOT] = nil
    if type(prev.unload) == "function" then
        local ok, err = pcall(prev.unload)
        if not ok then
            warn("Aftermath: previous instance unload failed: " .. tostring(err))
        end
        return true
    end
    return false
end

local hadPrevious = teardownPrevious()

local function attr(inst, name)
    if typeof(inst) ~= "Instance" then
        return nil
    end
    local ok, value = pcall(inst.GetAttribute, inst, name)
    if ok then
        return value
    end
    return nil
end

local function isCharacterModel(model)
    if typeof(model) ~= "Instance" or not model:IsA("Model") then
        return false
    end
    return model:FindFirstChild("Humanoid") ~= nil or model:FindFirstChildOfClass("AnimationController") ~= nil
end

local function rootOf(model)
    if typeof(model) ~= "Instance" then
        return nil
    end
    return model:FindFirstChild("HumanoidRootPart")
        or model:FindFirstChild("UpperTorso")
        or model:FindFirstChild("Torso")
        or model.PrimaryPart
        or model:FindFirstChildWhichIsA("BasePart")
end

local charCache = {}

local function characterOf(player)
    if player.Character and player.Character.Parent then
        charCache[player] = player.Character
        return player.Character
    end
    local cached = charCache[player]
    if cached and cached.Parent then
        return cached
    end
    if not Entities then
        return nil
    end
    for _, model in ipairs(Entities:GetChildren()) do
        if model:IsA("Model") and attr(model, "Player") == player then
            charCache[player] = model
            return model
        end
    end
    return nil
end

local function playerHealth(player)
    local h = attr(player, "Health")
    local m = attr(player, "MaxHealth")
    if h then
        return h, m or 100
    end
    local char = characterOf(player)
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        return hum.Health, hum.MaxHealth
    end
    return 0, 0
end

local function isDead(player)
    return attr(player, "Dead") == true
end

local function squadOf(player)
    return attr(player, "SquadName")
end

local function isSquadmate(player)
    local mine = squadOf(LP)
    if not mine then
        return false
    end
    return squadOf(player) == mine
end

local PLAYER_MODELS = {}

local function refreshPlayerModels()
    PLAYER_MODELS = {}
    for _, player in ipairs(Players:GetPlayers()) do
        local char = characterOf(player)
        if char then
            PLAYER_MODELS[char] = player
        end
    end
end

local function zombieModels()
    if not Entities then
        return {}
    end
    local out = {}
    for _, model in ipairs(Entities:GetChildren()) do
        if isCharacterModel(model) and not PLAYER_MODELS[model] then
            out[#out + 1] = model
        end
    end
    return out
end

local function lootModels()
    local out = {}
    local want = State.lootRarity
    local sources = { ItemSpawns }
    for _, folder in ipairs(sources) do
        if folder then
            for _, item in ipairs(folder:GetDescendants()) do
                if item:IsA("Model") then
                    local display = attr(item, "DisplayName")
                    if display then
                        local rarity = attr(item, "Rarity") or "Common"
                        if want == "All" or rarity == want then
                            out[#out + 1] = item
                        end
                    end
                end
            end
        end
    end
    return out
end

local function containerModels()
    local out = {}
    for _, model in ipairs(Workspace:GetChildren()) do
        if model:IsA("Model") and model.Name == "Container" then
            out[#out + 1] = model
        end
    end
    return out
end

local function vehicleModels()
    local out = {}
    for _, model in ipairs(Workspace:GetChildren()) do
        if model:IsA("Model") and model.Name == "WorldModel" then
            local states = model:FindFirstChild("States")
            if states then
                out[#out + 1] = model
            end
        end
    end
    return out
end

local function folderChildren(folder)
    if not folder then
        return {}
    end
    return folder:GetChildren()
end

local esp

local function setupESP()
    if type(ESP) ~= "table" or type(ESP.new) ~= "function" then
        return false
    end

    esp = ESP.new()
    esp:profile("high")

    local c = esp.cfg
    c.maxRange = 5000
    c.espRange = 5000
    c.target.on = false
    c.self.fov.on = false
    c.radar.on = false
    c.list.on = false
    c.threat.on = false

    c.npc.maxRange = 3000
    c.npc.espRange = 3000

    esp:setNPCSource(function()
        if not State.zombieOn then
            return {}
        end
        return zombieModels()
    end)

    esp:setNPCLabel(function(model)
        local health = attr(model, "Health")
        if health then
            return string.format("Infected [%d]", math.floor(health + 0.5))
        end
        return "Infected"
    end)

    esp:flag("squad", function(s)
        if s.plr and isSquadmate(s.plr) then
            return { "SQUAD", Color3.fromRGB(120, 190, 255) }
        end
        return nil
    end)

    esp:flag("status", function(s)
        if not s.plr then
            return nil
        end
        local bits = {}
        if attr(s.plr, "Bleeding") then
            bits[#bits + 1] = "BLEED"
        end
        if attr(s.plr, "BrokenLeg") then
            bits[#bits + 1] = "LEG"
        end
        if #bits > 0 then
            return { table.concat(bits, " "), Color3.fromRGB(255, 130, 140) }
        end
        return nil
    end)

    local function entityGroup(id, getter, labeler, colour, priority)
        return esp:addEnt(id, {
            get = getter,
            label = labeler,
            col = colour,
            box = true,
            name = true,
            dist = true,
            max = 5000,
            priority = priority,
        })
    end

    local lootSpec = entityGroup("loot", function()
        return State.lootOn and lootModels() or {}
    end, function(obj)
        local name = attr(obj, "DisplayName") or obj.Name
        local rarity = attr(obj, "Rarity")
        if rarity and rarity ~= "Common" then
            return name .. " [" .. rarity .. "]"
        end
        return name
    end, Color3.fromRGB(120, 255, 190), 2)

    State.lootSpec = lootSpec

    entityGroup("weapons", function()
        return State.weaponOn and folderChildren(GroundWeapons) or {}
    end, function(obj)
        return attr(obj, "DisplayName") or obj.Name
    end, Color3.fromRGB(255, 220, 110), 2)

    entityGroup("ammo", function()
        return State.ammoOn and folderChildren(RefillAmmo) or {}
    end, function()
        return "Ammo"
    end, Color3.fromRGB(255, 180, 120), 2)

    entityGroup("containers", function()
        return State.containerOn and containerModels() or {}
    end, function()
        return "Airdrop"
    end, Color3.fromRGB(255, 140, 90), 3)

    entityGroup("vehicles", function()
        return State.vehicleOn and vehicleModels() or {}
    end, function(obj)
        local states = obj:FindFirstChild("States")
        local hp = states and attr(states, "Health")
        if hp then
            return string.format("Vehicle [%d]", math.floor(hp + 0.5))
        end
        return "Vehicle"
    end, Color3.fromRGB(160, 200, 255), 3)

    if POIFolder then
        esp:addEnt("poi", {
            get = function()
                if not State.poiOn then
                    return {}
                end
                local out = {}
                for _, poi in ipairs(POIFolder:GetChildren()) do
                    local pos = attr(poi, "Position")
                    if typeof(pos) == "Vector3" then
                        out[#out + 1] = pos
                    end
                end
                return out
            end,
            label = false,
            col = Color3.fromRGB(200, 200, 210),
            box = false,
            name = false,
            dist = true,
            max = 0,
            priority = 1,
        })
    end

    esp:on(false)
    esp:start()
    return true
end

local aim

local function setupAim()
    if type(EasyAim) ~= "table" or type(EasyAim.new) ~= "function" then
        return false
    end

    aim = EasyAim.new({
        mode = "legit",
        key = Enum.UserInputType.MouseButton2,
        hitbox = { kind = "Head" },
        fov = { on = true, radius = 130, col = Color3.fromRGB(150, 220, 160) },
        legit = { smooth = 0.18, maxStep = 60 },
        range = 1500,
        deadCheck = true,
    })

    aim:setAdapter({
        targets = function()
            local out = {}
            if State.aimTargetPlayers then
                for _, player in ipairs(Players:GetPlayers()) do
                    if player ~= LP then
                        out[#out + 1] = player
                    end
                end
            end
            if State.aimTargetZombies then
                for _, model in ipairs(zombieModels()) do
                    out[#out + 1] = model
                end
            end
            return out
        end,
        model = function(raw)
            if typeof(raw) == "Instance" and raw:IsA("Player") then
                return characterOf(raw)
            end
            return raw
        end,
        root = rootOf,
        part = function(model, kind)
            if kind == "Head" then
                return model:FindFirstChild("Head")
            elseif kind == "Torso" then
                return model:FindFirstChild("UpperTorso")
                    or model:FindFirstChild("Torso")
                    or model:FindFirstChild("HumanoidRootPart")
            end
            return nil
        end,
        health = function(model, raw)
            if typeof(raw) == "Instance" and raw:IsA("Player") then
                return playerHealth(raw)
            end
            local h = attr(model, "Health")
            if h then
                return h, attr(model, "MaxHealth") or 100
            end
            local hum = model and model:FindFirstChildOfClass("Humanoid")
            if hum then
                return hum.Health, hum.MaxHealth
            end
            return 1, 1
        end,
        alive = function(model, raw, health, maxHealth)
            if typeof(raw) == "Instance" and raw:IsA("Player") and isDead(raw) then
                return false
            end
            if maxHealth and maxHealth > 0 then
                return health > 0
            end
            return model ~= nil
        end,
        name = function(model, raw)
            if typeof(raw) == "Instance" and raw:IsA("Player") then
                return raw.DisplayName
            end
            return "Infected"
        end,
        team = function(model, raw)
            if typeof(raw) == "Instance" and raw:IsA("Player") then
                return squadOf(raw)
            end
            return nil
        end,
        ignore = function(model, raw)
            return raw == LP
        end,
    })

    aim:setFilter(function(rec)
        if typeof(rec.raw) == "Instance" and rec.raw:IsA("Player") then
            if isSquadmate(rec.raw) then
                return false
            end
        end
        return true
    end)

    aim:on(false)
    aim:start()
    return true
end

M:Every("aftermath:refresh", 1, function()
    refreshPlayerModels()

    local health, maxHealth = playerHealth(LP)
    if maxHealth and maxHealth > 0 then
        labelHealth.Text = string.format("Health: %d / %d", math.floor(health + 0.5), math.floor(maxHealth + 0.5))
    else
        labelHealth.Text = "Health: -"
    end

    local hunger = attr(LP, "Hunger")
    local maxHunger = attr(LP, "MaxHunger") or 100
    labelHunger.Text = hunger and string.format("Hunger: %d / %d", math.floor(hunger + 0.5), maxHunger) or "Hunger: -"

    local thirst = attr(LP, "Thirst")
    local maxThirst = attr(LP, "MaxThirst") or 100
    labelThirst.Text = thirst and string.format("Thirst: %d / %d", math.floor(thirst + 0.5), maxThirst) or "Thirst: -"

    local stamina = attr(LP, "Stamina")
    local maxStamina = attr(LP, "MaxStamina") or 100
    labelStamina.Text = stamina and string.format("Stamina: %d / %d", math.floor(stamina + 0.5), maxStamina) or "Stamina: -"

    local status = {}
    if attr(LP, "Bleeding") then
        status[#status + 1] = "Bleeding"
    end
    if attr(LP, "BrokenLeg") then
        status[#status + 1] = "Broken Leg"
    end
    local infection = attr(LP, "Infection")
    if infection and infection > 0 then
        status[#status + 1] = string.format("Infection %d", math.floor(infection + 0.5))
    end
    local toxicity = attr(LP, "Toxicity")
    if toxicity and toxicity > 0 then
        status[#status + 1] = string.format("Toxicity %d", math.floor(toxicity + 0.5))
    end
    labelStatus.Text = #status > 0 and ("Status: " .. table.concat(status, ", ")) or "Status: healthy"

    local squad = squadOf(LP)
    if squad then
        local mates = 0
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LP and squadOf(player) == squad then
                mates = mates + 1
            end
        end
        labelSquad.Text = string.format("Squad: %s (%d mates)", squad, mates)
    else
        labelSquad.Text = "Squad: none"
    end

    local serverName = GameAssets and GameAssets:FindFirstChild("server_name")
    local regionName = GameAssets and GameAssets:FindFirstChild("server_region_name")
    labelServer.Text = string.format(
        "Server: %s (%s)",
        serverName and serverName.Value or "?",
        regionName and regionName.Value or "?"
    )
    labelPlayers.Text = string.format("Players: %d / %d", #Players:GetPlayers(), Players.MaxPlayers)
end, 5)

function Instance_.unload()
    for _, conn in ipairs(Instance_.conns) do
        pcall(function()
            conn:Disconnect()
        end)
    end
    Instance_.conns = {}

    if aim then
        pcall(function()
            aim:destroy()
        end)
    end
    if esp then
        pcall(function()
            esp:destroy()
        end)
    end

    local light = Instance_.restore.lighting
    if light then
        pcall(function()
            Lighting.Brightness = light.Brightness
            Lighting.ClockTime = light.ClockTime
            Lighting.FogEnd = light.FogEnd
            Lighting.Ambient = light.Ambient
            Lighting.OutdoorAmbient = light.OutdoorAmbient
        end)
    end

    pcall(function()
        local cam = Workspace.CurrentCamera
        if cam and Instance_.restore.fov then
            cam.FieldOfView = Instance_.restore.fov
        end
    end)

    pcall(function()
        M:Close()
    end)

    if GENV[SLOT] == Instance_ then
        GENV[SLOT] = nil
    end
end
